// Helpers
//------------------------------------------------------------------------------
f_median3(array<int> a) =>
    float out = na
    if array.size(a) >= 3
        float x = float(array.get(a, 0))
        float y = float(array.get(a, 1))
        float z = float(array.get(a, 2))
        out := x + y + z - math.max(x, math.max(y, z)) - math.min(x, math.min(y, z))
    out

f_push3(array<int> a, int v) =>
    array.push(a, v)
    if array.size(a) > 3
        array.shift(a)

f_cosPrice(bool bull, float H, float L, float angleDeg) =>
    float r = H - L
    float c = math.cos(math.toradians(angleDeg))
    bull ? H - r * c : L + r * c

f_sqLevel(float anchor, float step, float angleDeg, bool up) =>
    float base = math.sqrt(math.max(anchor, 0.0))
    float inc  = step * angleDeg / 360.0
    float root = up ? base + inc : base - inc
    root > 0 ? root * root : na

f_fmt(float x) => str.tostring(x, format.mintick)
f_fmt1(float x) => str.tostring(x, "#.0")
f_fmt2(float x) => str.tostring(x, "#.00")

f_crossCycleAngle(float currentTotal, float previousTotal, float targetDeg) =>
    bool crossed = false
    if not na(currentTotal) and not na(previousTotal)
        crossed := math.floor((currentTotal - targetDeg) / 360.0) > math.floor((previousTotal - targetDeg) / 360.0)
    crossed

f_waveStage(float phaseDeg) =>
    string stage = "Mean / cycle boundary"
    if phaseDeg >= 30.0 and phaseDeg < 90.0
        stage := "Rising positive half"
    else if phaseDeg >= 90.0 and phaseDeg < 150.0
        stage := "Falling from +1 crest"
    else if phaseDeg >= 150.0 and phaseDeg < 210.0
        stage := "Mean crossing zone"
    else if phaseDeg >= 210.0 and phaseDeg < 270.0
        stage := "Falling negative half"
    else if phaseDeg >= 270.0 and phaseDeg < 330.0
        stage := "Rising from -1 trough"
    stage

f_clearPolylines(array<polyline> ids) =>
    while array.size(ids) > 0
        polyline.delete(array.pop(ids))

f_storeCurvedPolyline(array<chart.point> points, array<polyline> ids, color lineColor, int lineWidth, bool overlayDrawing, int maxSegments) =>
    if array.size(points) >= 2
        polyline curve = polyline.new(
             points = points,
             curved = true,
             closed = false,
             xloc = xloc.bar_index,
             line_color = lineColor,
             fill_color = na,
             line_style = line.style_solid,
             line_width = lineWidth,
             force_overlay = overlayDrawing
         )
        array.push(ids, curve)
        if array.size(ids) > maxSegments
            polyline.delete(array.shift(ids))

float cycleProgress = haveSwing ? 100.0 * barsInCycle / cycleLen : na

// SIN phase model

//       SIN(theta) = Mean + Amplitude * sin(Time Phase).
//       It is a phase model, not a standalone entry signal and not the COS retracement map.

float nearest30Raw = haveSwing ? math.round(timePhase / 30.0) * 30.0 : na
float nearest30Wrapped = haveSwing ? (nearest30Raw >= 360.0 ? 0.0 : nearest30Raw) : na
float nearest30DiffRaw = haveSwing ? math.abs(timePhase - nearest30Wrapped) : na
float nearest30Diff = haveSwing ? math.min(nearest30DiffRaw, 360.0 - nearest30DiffRaw) : na
float keyPointTolerance = haveSwing ? math.min(15.0, math.max(1.0, degPerBar / 2.0)) : na
bool waveKeyPoint = haveSwing and nearest30Diff <= keyPointTolerance
float waveKeyPointY = waveKeyPoint ? math.sin(math.toradians(nearest30Raw)) : na

float cosGeometryWave = haveSwing ? f_cosPrice(bull, swingH, swingL, timePhase) : na
float geometryDelta = haveSwing ? close - cosGeometryWave : na
float geometryDistancePct = haveSwing and swingRange > 0 ? 100.0 * math.abs(geometryDelta) / swingRange : na
bool geometryNear = haveSwing and geometryDistancePct <= tolerancePct

float cos45 = haveSwing ? f_cosPrice(bull, swingH, swingL, 45.0) : na
float cos60 = haveSwing ? f_cosPrice(bull, swingH, swingL, 60.0) : na
float zoneLow  = haveSwing ? math.min(cos30, cos60) : na
float zoneHigh = haveSwing ? math.max(cos30, cos60) : na
float tol = haveSwing ? math.max(syminfo.mintick, swingRange * tolerancePct / 100.0) : na
bool inZone = haveSwing and close >= zoneLow and close <= zoneHigh
float zoneDist = haveSwing ? (close < zoneLow ? zoneLow - close : close > zoneHigh ? close - zoneHigh : 0.0) : na
bool nearZone = haveSwing and (inZone or zoneDist <= tol)

// Square of 9 proximity
//------------------------------------------------------------------------------
var array<float> sqAngles = array.from(45.0, 90.0, 120.0, 135.0, 180.0, 225.0, 270.0, 315.0, 360.0)
float sqNearDist = 1e20
float sqNearPrice = na
float sqNearAngle = na
if haveSwing
    for i = 0 to array.size(sqAngles) - 1
        float a = array.get(sqAngles, i)
        float up = f_sqLevel(anchorPrice, sq9Step, a, true)
        float dn = f_sqLevel(anchorPrice, sq9Step, a, false)
        if not na(up) and math.abs(close - up) < sqNearDist
            sqNearDist := math.abs(close - up)
            sqNearPrice := up
            sqNearAngle := a
        if not na(dn) and math.abs(close - dn) < sqNearDist
            sqNearDist := math.abs(close - dn)
            sqNearPrice := dn
            sqNearAngle := -a
int sqPts = haveSwing and sqNearDist <= tol ? 1 : 0

//------------------------------------------------------------------------------
// Gann fan proximity
//------------------------------------------------------------------------------
var array<float> gannRatios = array.from(0.125, 0.25, 0.5, 1.0, 2.0, 4.0, 8.0)
float sign = bull ? 1.0 : -1.0
float gannNearDist = 1e20
float gannNearPrice = na
float gannNearRatio = na
if haveSwing
    for i = 0 to array.size(gannRatios) - 1
        float ratio = array.get(gannRatios, i)
        float gp = anchorPrice + sign * gannScale * ratio * barsSince
        float gd = math.abs(close - gp)
        if gd < gannNearDist
            gannNearDist := gd
            gannNearPrice := gp
            gannNearRatio := ratio
int gannPts = haveSwing and gannNearDist <= tol ? 1 : 0

float priorHigh = ta.highest(high, bosLen)[1]
float priorLow  = ta.lowest(low, bosLen)[1]
bool bosBullRaw = haveSwing and not na(priorHigh) and close > priorHigh
bool bosBearRaw = haveSwing and not na(priorLow) and close < priorLow
bool bosRaw = bull ? bosBullRaw : bosBearRaw
bool bosConfirmed = bosRaw and (not requireBarClose or barstate.isconfirmed)

// Optional research values. They are displayed but are not mandatory gates.
bool rejectionBull = haveSwing and low <= zoneHigh and close > open and close >= cos45
bool rejectionBear = haveSwing and high >= zoneLow and close < open and close <= cos45
bool rejection = bull ? rejectionBull : rejectionBear
float rsi = ta.rsi(close, rsiLen)
bool momentum = bull ? rsi > 50.0 : rsi < 50.0

// Mandatory WHERE and WHEN gates.
bool priceGateNow = nearZone
bool timeWindow = haveSwing and timeDiff <= fullWindowDeg
bool approachingWindow = haveSwing and not timeWindow and timeDiff <= approachDeg

// Mandatory risk gate. READY uses RR from the actual trigger bar close to the original swing target.
float preferredEntry = cos45
float stopPrice = haveSwing ? (bull ? swingL - slBuffer : swingH + slBuffer) : na
float swingTarget = haveSwing ? (bull ? swingH : swingL) : na
float currentRisk = haveSwing ? math.abs(close - stopPrice) : na
float currentReward = haveSwing ? (bull ? swingTarget - close : close - swingTarget) : na
float currentRR = haveSwing and currentRisk > 0.0 and currentReward > 0.0 ? currentReward / currentRisk : 0.0
bool riskGate = haveSwing and currentReward > 0.0 and currentRR >= minRR

float plannedRisk = haveSwing ? math.abs(preferredEntry - stopPrice) : na
float plannedReward = haveSwing ? (bull ? swingTarget - preferredEntry : preferredEntry - swingTarget) : na
float plannedRR = haveSwing and plannedRisk > 0.0 and plannedReward > 0.0 ? plannedReward / plannedRisk : na

// Invalidation is based on a close beyond the anchor swing extreme.
bool invalidated = haveSwing and (bull ? close < swingL : close > swingH)
int biasCode = bull ? 1 : -1

// ARMED memory: Price + Time may occur before BOS. Once armed, the setup can wait
// for a confirmed structure break for a limited number of bars.
var bool setupArmed = false
var int armedBias = 0
var int armedAtBar = na
var int trackedAnchorIdx = na
var int trackedBias = 0

// Drawing state for the clear cycle-wave pane.
var array<label> waveAngleLabelIds = array.new<label>()
var label currentPhaseLabel = na
var line currentPhaseLine = na
var array<float> waveLabelAngles = array.from(30.0, 60.0, 90.0, 120.0, 150.0, 180.0, 210.0, 240.0, 270.0, 300.0, 330.0, 360.0)
var array<polyline> smoothSinPolylineIds = array.new<polyline>()
var array<polyline> smoothCosPolylineIds = array.new<polyline>()

bool anchorChanged = haveSwing and (na(trackedAnchorIdx) or anchorIdx != trackedAnchorIdx or biasCode != trackedBias)
if anchorChanged
    setupArmed := false
    armedBias := 0
    armedAtBar := na
    trackedAnchorIdx := anchorIdx
    trackedBias := biasCode
    while array.size(waveAngleLabelIds) > 0
        label.delete(array.pop(waveAngleLabelIds))
    if showCycleWave and showWaveAngleLabels and not na(anchorIdx)
        label anchorLabel = label.new(anchorIdx, 0.0, "0°", xloc = xloc.bar_index, yloc = yloc.price, style = label.style_none, textcolor = color.new(color.gray, 0), size = size.tiny)
        array.push(waveAngleLabelIds, anchorLabel)

bool armExpired = setupArmed and not na(armedAtBar) and bar_index - armedAtBar > armExpiryBars
if invalidated or armExpired
    setupArmed := false
    armedBias := 0
    armedAtBar := na

bool armCondition = haveSwing and not invalidated and priceGateNow and timeWindow
bool armPulse = false
if armCondition and not setupArmed
    setupArmed := true
    armedBias := biasCode
    armedAtBar := bar_index
    armPulse := true

bool armedForEvaluation = setupArmed and armedBias == biasCode
bool readyPulse = armedForEvaluation and bosConfirmed and riskGate and not invalidated
bool poorRRBlocked = armedForEvaluation and bosConfirmed and not riskGate and not invalidated

// The two setup gates remain locked after ARMED, so BOS can occur after price leaves the zone.
bool whereGate = armedForEvaluation or priceGateNow
bool whenGate = armedForEvaluation or timeWindow
int coreGatePass = (whereGate ? 1 : 0) + (whenGate ? 1 : 0) + (bosConfirmed ? 1 : 0) + (riskGate ? 1 : 0)
float armedAge = armedForEvaluation and not na(armedAtBar) ? float(bar_index - armedAtBar) : na
float armedBarsLeft = armedForEvaluation and not na(armedAtBar) ? math.max(0.0, float(armExpiryBars - (bar_index - armedAtBar))) : na

int statusCode = 0
string entryStatus = "NO TRADE"
string entryReason = "Waiting for Price + Time alignment"
if not haveSwing
    statusCode := 0
    entryStatus := "WAITING PIVOTS"
    entryReason := "Need confirmed Swing High and Swing Low"
else if invalidated
    statusCode := 0
    entryStatus := "INVALID"
    entryReason := bull ? "Close broke below Swing Low" : "Close broke above Swing High"
else if readyPulse
    statusCode := bull ? 3 : -3
    entryStatus := bull ? "BUY READY" : "SELL READY"
    entryReason := "WHERE + WHEN + BOS + RR all passed"
else if poorRRBlocked
    statusCode := 0
    entryStatus := "NO TRADE - POOR RR"
    entryReason := "BOS confirmed, but RR to swing target is below minimum"
else if armedForEvaluation
    statusCode := bull ? 2 : -2
    entryStatus := bull ? "ARMED BUY" : "ARMED SELL"
    entryReason := "Price + Time locked; waiting for confirmed-close BOS"
else if priceGateNow and approachingWindow
    statusCode := bull ? 1 : -1
    entryStatus := bull ? "WAIT TIME BUY" : "WAIT TIME SELL"
    entryReason := "Price is ready; Time Window is approaching"
else if priceGateNow and not timeWindow
    statusCode := bull ? 1 : -1
    entryStatus := bull ? "WAIT TIME BUY" : "WAIT TIME SELL"
    entryReason := "Price is in/near COS zone; Time Window is not active"
else if timeWindow and not priceGateNow
    statusCode := bull ? 1 : -1
    entryStatus := bull ? "WAIT PRICE BUY" : "WAIT PRICE SELL"
    entryReason := "Time Window is active; price has not reached COS zone"
else
    statusCode := 0
    entryStatus := "NO TRADE"
    entryReason := approachingWindow ? "Time is approaching, but price is outside the COS zone" : "Price and Time are not aligned"

// A READY signal or a poor-RR block completes this armed attempt.
if readyPulse or poorRRBlocked
    setupArmed := false
    armedBias := 0
    armedAtBar := na

//------------------------------------------------------------------------------
// Cycle-wave labels and current phase marker
//------------------------------------------------------------------------------
if barstate.isnew and showCycleWave and showWaveAngleLabels and haveSwing and not na(totalAngle[1])
    for i = 0 to array.size(waveLabelAngles) - 1
        float angleValue = array.get(waveLabelAngles, i)
        if f_crossCycleAngle(totalAngle, totalAngle[1], angleValue)
            float labelY = math.sin(math.toradians(angleValue))
            string angleText = angleValue == 360.0 ? "360° / 0°" : str.tostring(int(angleValue)) + "°"
            label angleLabel = label.new(bar_index, labelY, angleText, xloc = xloc.bar_index, yloc = yloc.price, style = label.style_none, textcolor = color.new(color.gray, 5), size = size.tiny)
            array.push(waveAngleLabelIds, angleLabel)
            if array.size(waveAngleLabelIds) > 37
                label.delete(array.shift(waveAngleLabelIds))

if barstate.islast
    if not showWaveAngleLabels
        while array.size(waveAngleLabelIds) > 0
            label.delete(array.pop(waveAngleLabelIds))
    if not na(currentPhaseLabel)
        label.delete(currentPhaseLabel)
        currentPhaseLabel := na
    if not na(currentPhaseLine)
        line.delete(currentPhaseLine)
        currentPhaseLine := na
    if showCycleWave and showCurrentPhase and haveSwing
        currentPhaseLine := line.new(bar_index, -1.08, bar_index, 1.08, xloc = xloc.bar_index, extend = extend.none, color = color.new(color.red, 15), style = line.style_dashed, width = 1)
        string phaseText = "NOW  " + f_fmt1(timePhase) + "°\nSIN  " + f_fmt2(sinCycleNorm) + "\n" + f_waveStage(timePhase)
        currentPhaseLabel := label.new(bar_index, sinCycleNorm, phaseText, xloc = xloc.bar_index, yloc = yloc.price, style = label.style_label_left, color = color.new(color.red, 15), textcolor = color.white, size = size.small)

//------------------------------------------------------------------------------
// Smooth visual curves
//------------------------------------------------------------------------------
// Pine plots provide the exact series values used by Crosshair/Data Window.
// Curved polylines are added only for visual interpolation and are split whenever
// the confirmed swing anchor or bias changes, so a curve cannot bend across a reset.
if barstate.islast
    f_clearPolylines(smoothSinPolylineIds)
    f_clearPolylines(smoothCosPolylineIds)
    if haveSwing and ((showCycleWave and smoothSinCurve) or (showGeometryMap and showCosGeometry and smoothCosCurve))
        int historyBars = math.min(smoothHistoryBars - 1, bar_index)
        array<chart.point> sinPoints = array.new<chart.point>()
        array<chart.point> cosPoints = array.new<chart.point>()
        int segmentAnchor = na
        int segmentBias = 0
        for scan = 0 to historyBars
            int historyOffset = historyBars - scan
            bool validPoint = not na(anchorIdx[historyOffset]) and not na(sinCycleNorm[historyOffset]) and not na(cosGeometryWave[historyOffset])
            int pointAnchor = validPoint ? anchorIdx[historyOffset] : na
            int pointBias = validPoint ? (bull[historyOffset] ? 1 : -1) : 0
            bool segmentBreak = validPoint and not na(segmentAnchor) and (pointAnchor != segmentAnchor or pointBias != segmentBias)
            if not validPoint or segmentBreak
                if showCycleWave and smoothSinCurve
                    f_storeCurvedPolyline(sinPoints, smoothSinPolylineIds, sinWaveColor, sinWaveWidth, false, smoothMaxSegments)
                if showGeometryMap and showCosGeometry and smoothCosCurve
                    f_storeCurvedPolyline(cosPoints, smoothCosPolylineIds, geometryColor, geometryWidth, true, smoothMaxSegments)
                sinPoints := array.new<chart.point>()
                cosPoints := array.new<chart.point>()
                segmentAnchor := na
                segmentBias := 0
            if validPoint
                int pointIndex = bar_index - historyOffset
                if showCycleWave and smoothSinCurve
                    array.push(sinPoints, chart.point.from_index(pointIndex, sinCycleNorm[historyOffset]))
                if showGeometryMap and showCosGeometry and smoothCosCurve
                    array.push(cosPoints, chart.point.from_index(pointIndex, cosGeometryWave[historyOffset]))
